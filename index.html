function processMatches() { const input = document.getElementById('input').value.trim().split('\n').map(line => line.trim()).filter(line => line); let output = ''; let matchCount = 0; let teamName = ''; // Přidáno pro uložení jména týmu nebo hráče // Získání jména týmu nebo hráče z prvního řádku if (input.length > 0) { teamName = input[0]; } for (let i = 1; i < input.length; ) { // Změněn index na 1, abychom začali od druhého řádku if (input[i].startsWith('Last matches:')) { i++; continue; } if (i + 6 > input.length) break; const [date, tournament, player, opponent, score1, score2, result] = input.slice(i, i + 7); // Check if the match is a "premature end" match if (score1 === '-' && score2 === '-' && result === 'W') { output += `${player} vs ${opponent}: předčasné ukončení - Výhra\n`; matchCount++; i += 7; continue; } const parsedScore1 = parseInt(score1, 10); const parsedScore2 = parseInt(score2, 10); if (isNaN(parsedScore1) || isNaN(parsedScore2)) { i++; continue; } let matchResult; if (parsedScore1 === parsedScore2) { matchResult = 'Remíza'; } else { matchResult = result === 'W' ? 'Výhra' : 'Prohra'; } output += `${player} vs ${opponent}: ${parsedScore1}-${parsedScore2} - ${matchResult}\n`; matchCount++; i += 7; } // Uložení výsledku do zmiennej lastOutput a zobrazení na stránce lastOutput = `Analyzovaný tým/hráč: ${teamName}\nCelkový počet zápasů: ${matchCount}\n\n${output.trim()}`; document.getElementById('output').textContent = lastOutput; }
